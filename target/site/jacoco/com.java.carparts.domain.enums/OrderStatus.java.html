package com.java.carparts.domain.enums;

/**
 * PLACED is the initial state. From there:
 *   - CONFIRMED or DECLINED (DECLINED is terminal - never moves forward)
 *   - CONFIRMED -> IN_PROCESS -> DELIVERY -> COMPLETED
 *
 * NOTE: only PLACED -> CONFIRMED is actually wired up right now (see
 * OrderService). The rest of the lifecycle is defined here so the shape
 * exists, but the transition logic itself is intentionally deferred -
 * this becomes real business logic in a later lecture, not today.
 */
public enum OrderStatus {
    PLACED,
    CONFIRMED,
    DECLINED,
    IN_PROCESS,
    DELIVERY,
    COMPLETED
}
