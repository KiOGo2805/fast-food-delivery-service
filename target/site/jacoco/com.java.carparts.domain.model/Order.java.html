package com.java.carparts.domain.model;

import com.java.carparts.domain.enums.OrderStatus;
import jakarta.persistence.*;
import lombok.AllArgsConstructor;
import lombok.Getter;
import lombok.NoArgsConstructor;
import lombok.Setter;

import java.math.BigDecimal;
import java.time.OffsetDateTime;
import java.util.ArrayList;
import java.util.List;

@Entity
@Getter
@Setter
@Table(name = "orders")
@NoArgsConstructor
@AllArgsConstructor
public class Order {

    @Id
    @GeneratedValue(strategy = GenerationType.IDENTITY)
    private Integer id;

    @OneToMany(mappedBy = "order", cascade = CascadeType.ALL, orphanRemoval = true)
    private List<OrderItem> items = new ArrayList<>();

    // Stored, not computed on the fly - the total is calculated once at
    // order time (from the item snapshots) and kept as-is afterward, same
    // reasoning as OrderItem.unitPrice: order history should stay accurate
    // to what was actually charged, not drift if item data changes later.
    @Column(name = "total_price", nullable = false)
    private BigDecimal totalPrice;

    @Enumerated(EnumType.STRING)
    @Column(nullable = false)
    private OrderStatus status;

    @Column(name = "customer_username", nullable = false)
    private String customerUsername;

    @Column(name = "created_at", nullable = false)
    private OffsetDateTime createdAt;

    // Keeps both sides of the relationship in sync in one call - a common,
    // small but easy-to-forget JPA convenience method for bidirectional
    // associations.
    public void addItem(OrderItem item) {
        items.add(item);
        item.setOrder(this);
    }
}
